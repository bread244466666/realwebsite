server.js (Node.js + Express + Mongoose) 
// Express Server & MongoDB Connection (process.env.MONGODB_URI)
const express = require('express');
const mongoose = require('mongoose');
const cors = require('cors');

const app = express();
app.use(cors());
app.use(express.json({ limit: '10mb' }));

// Connect to MongoDB using environment variable
const MONGODB_URI = process.env.MONGODB_URI;

mongoose.connect(MONGODB_URI)
  .then(() => console.log('Successfully connected to MongoDB Database'))
  .catch(err => console.error('MongoDB Connection Error:', err));

// Postcard Schema
const postcardSchema = new mongoose.Schema({
  title: { type: String, required: true },
  source: { type: String, enum: ['Flower', 'Mushroom Fight'], required: true },
  location: { type: String, required: true },
  date: { type: String, required: true },
  imageUrl: { type: String, required: true },
  createdAt: { type: Date, default: Date.now }
});

const Postcard = mongoose.model('Postcard', postcardSchema);

// GET API Endpoint
app.get('/api/postcards', async (req, res) => {
  try {
    const postcards = await Postcard.find().sort({ createdAt: -1 });
    res.json(postcards);
  } catch (err) {
    res.status(500).json({ error: 'Failed to fetch postcards' });
  }
});

// POST API Endpoint
app.post('/api/postcards', async (req, res) => {
  try {
    const { title, source, location, date, imageUrl } = req.body;
    const newPostcard = new Postcard({ title, source, location, date, imageUrl });
    await newPostcard.save();
    res.status(201).json(newPostcard);
  } catch (err) {
    res.status(400).json({ error: 'Invalid postcard data' });
  }
});

const PORT = process.env.PORT || 5000;
app.listen(PORT, () => console.log(`Server listening on port ${PORT}`));
Connects using environment variable process.env.MONGODB_URI
